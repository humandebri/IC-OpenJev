import {existsSync,mkdirSync,readFileSync,writeFileSync,copyFileSync} from 'node:fs';
import {spawn} from 'node:child_process';
import {createInterface} from 'node:readline';
import {connectQuery,validateReply,type ModelReply} from '../src/placement-query-api';
import {POLICIES,TEST_SEEDS,VALID_SEEDS,studyGame,summarize,adoption,hash,loadEvaluator,type Answer,type Decide} from './placement-study-core';
const out='artifacts/tetris-search';
const test=process.argv.includes('--test'),native=process.argv.includes('--native');
if(test&&native)throw new Error('Held-out evaluation requires actual canister replies');
const phase=test?'test':'validation';
const target=process.argv.includes('--local')?JSON.parse(readFileSync(`${out}/local-target.json`,'utf8')):{host:process.env.IC_HOST||'https://icp-api.io',canister:process.env.CANISTER_ID||'qojfj-6qaaa-aaaam-qjkaq-cai'};
if(test&&(!existsSync(`${out}/validation-summary.json`)||!JSON.parse(readFileSync(`${out}/validation-summary.json`,'utf8')).complete))throw new Error('Finish validation before revealing test results');
mkdirSync(`${out}/${phase}`,{recursive:true});mkdirSync(`${out}/query-cache`,{recursive:true});
const evaluator=loadEvaluator(`${out}/evaluator.json`);
const sourceFiles=['web/tetris/src/placement-local.ts','web/tetris/src/placement-search.ts','web/tetris/src/placement-query-api.ts','web/tetris/scripts/placement-study-core.ts','web/tetris/scripts/placement-study.ts'];
const sourceHashes=Object.fromEntries(sourceFiles.map(f=>[f,hash(readFileSync(f))]));
const protocol={phase,native,target:native?null:target,checkpoint_sha256:hash(readFileSync('models/verdict-pack/model.bin')),tokenizer_sha256:hash(readFileSync('models/verdict-151m/tokenizer.json')),probe_sha256:native?hash(readFileSync('target/release/tetris_probe')):null,seeds:test?TEST_SEEDS:VALID_SEEDS,policies:POLICIES,evaluator_hash:evaluator.hash,source_hashes:sourceHashes,model_digest:'0757d774c8c44397d189c5b479853897acef8f1e2b55ebf1f5a46969063225e0',gate:'10%-lines-nonregressing-survival-and-early-rate-v1',max_turns:128};
const protocolBytes=JSON.stringify(protocol,null,2)+'\n',protocolPath=`${out}/${phase}/protocol.json`;
if(existsSync(protocolPath)&&readFileSync(protocolPath,'utf8')!==protocolBytes)throw new Error('Protocol changed after evaluation began; keep prior evidence and use a new experiment directory');
writeFileSync(protocolPath,protocolBytes);
let actualQueries=0,cachedQueries=0;
let decide:Decide,close=()=>{};
if(native){
 const p=spawn('target/release/tetris_probe',['.'],{stdio:['pipe','pipe','inherit']});
 const lines=createInterface({input:p.stdout});const queue:((line:string)=>void)[]=[];
 lines.on('line',line=>{const next=queue.shift();if(!next)throw new Error('Unexpected probe output');next(line);});
 const cache=new Map<string,Answer>();
 decide=async request=>{
  const key=JSON.stringify(request);const saved=cache.get(key);if(saved)return {...saved,cached:true};
  const raw=await new Promise<string>((resolve,reject)=>{queue.push(resolve);p.stdin.write(JSON.stringify({text:request.state,labels:request.options.map(o=>o.text),infer:true})+'\n');});
  const r=JSON.parse(raw),answer={selected:r.selected,tokens:r.tokens,instructions:'0',cached:false};cache.set(key,answer);return answer;
 };close=()=>{p.stdin.end();};
}else{
 const {host,canister}=target;
 const client=await connectQuery(host,canister),status=await client.status();
 if(!status.enabled||Buffer.from(status.model).toString('hex')!==protocol.model_digest)throw new Error('Frozen model unavailable or changed');
 const nativeFetch=globalThis.fetch;
 globalThis.fetch=(...args)=>{
  const url=String(args[0] instanceof Request?args[0].url:args[0]);if(/\/call(?:\?|$)/.test(url))throw new Error('Update calls forbidden');return nativeFetch(...args);
 };
 decide=async request=>{
  const key=hash(JSON.stringify({request,model:protocol.model_digest,host,canister})),path=`${out}/query-cache/${key}.json`;
  const cached=existsSync(path);
  const reply:ModelReply=cached?JSON.parse(readFileSync(path,'utf8'),(key,value)=>key==='measured_instructions'?BigInt(value):value):await client.query(request);
  let selected:number;try{selected=validateReply(reply,request,status);}catch(error){writeFileSync(`${out}/${phase}/invalid-reply.json`,JSON.stringify({request,status,reply},(_,v)=>typeof v==='bigint'?v.toString():v,2)+'\n');throw error;}
  if(reply.input_tokens>41)throw new Error('Input exceeds frozen 41-token bound');
  if(!cached){writeFileSync(path,JSON.stringify({...reply,model:Array.from(reply.model)},(_,v)=>typeof v==='bigint'?v.toString():v)+'\n');actualQueries++;}else cachedQueries++;
  return {selected,tokens:reply.input_tokens,instructions:reply.measured_instructions.toString(),cached};
 };
}
const provider=decide;
const pending=new Map<string,Promise<Answer>>();
decide=async request=>{
 const key=JSON.stringify(request),saved=pending.get(key);
 if(saved){cachedQueries++;return {...await saved,cached:true};}
 const promise=provider(request);pending.set(key,promise);
 try{return await promise;}catch(error){pending.delete(key);throw error;}
};
const games:Awaited<ReturnType<typeof studyGame>>[]=[];
const jobs=protocol.seeds.flatMap((seed:number)=>POLICIES.map(policy=>({seed,policy})));
let cursor=0;
try{
 await Promise.all(Array.from({length:4},async()=>{while(cursor<jobs.length){
  const {seed,policy}=jobs[cursor++];
  const path=`${out}/${phase}/${seed}-${policy}.json`;
  const game=existsSync(path)?JSON.parse(readFileSync(path,'utf8')):await studyGame(seed,policy,evaluator,decide);
  if(!existsSync(path))writeFileSync(path,JSON.stringify(game)+'\n');games.push(game);games.sort((a,b)=>a.seed-b.seed||POLICIES.indexOf(a.policy)-POLICIES.indexOf(b.policy));
  const summary=summarize(games),complete=games.length===protocol.seeds.length*POLICIES.length;
  const result={protocol_sha256:hash(protocolBytes),complete,authoritative:!native,query_errors:0,budget_violations:0,actual_queries_this_run:actualQueries,cached_queries_this_run:cachedQueries,summary,adoption:test&&complete?adoption(summary,!native):null,
   games:games.map(({trace,...g})=>g),sources:games.map(g=>({path:`${phase}/${g.seed}-${g.policy}.json`,sha256:hash(readFileSync(`${out}/${phase}/${g.seed}-${g.policy}.json`))}))};
  writeFileSync(`${out}/${phase}-summary.json`,JSON.stringify(result,null,2)+'\n');
  console.log(`${phase} ${games.length}/${protocol.seeds.length*POLICIES.length}: ${seed} ${policy}: ${game.lines} lines / ${game.pieces} pieces; ${actualQueries} actual queries`);
  if(test&&complete){
   if(!existsSync(`${out}/validation-summary.json`)||!JSON.parse(readFileSync(`${out}/validation-summary.json`,'utf8')).complete)throw new Error('Validation must finish before adoption');
   writeFileSync('web/tetris/src/placement-release.json',JSON.stringify({...result.adoption,evidence_sha256:hash(readFileSync(`${out}/${phase}-summary.json`)),evaluator_hash:evaluator.hash},null,2)+'\n');
  }
 }}));
}catch(error){writeFileSync(`${out}/${phase}/failure.json`,JSON.stringify({error:String(error),completed_games:games.length},null,2)+'\n');throw error;}finally{close();}
